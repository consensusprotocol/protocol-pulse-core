    function initAudioContext() {
        if (!audioContext && audioElement) {
            audioContext = new (window.AudioContext || window.webkitAudioContext)();
            analyser = audioContext.createAnalyser();
            analyser.fftSize = 2048;
            analyser.smoothingTimeConstant = 0.85;

            audioSource.connect(analyser);
            analyser.connect(audioContext.destination);

    function resizeOscilloscope() {
        if (oscilloscopeCanvas) {
            const container = oscilloscopeCanvas.parentElement;
            oscilloscopeCanvas.width = container.offsetWidth;
            oscilloscopeCanvas.height = container.offsetHeight;
        }
    }
    resizeOscilloscope();

    function drawOscilloscope() {
        if (!oscilloscopeCanvas || !oscCtx) return;
        
        const width = oscilloscopeCanvas.width;
        const height = oscilloscopeCanvas.height;
        
        oscCtx.clearRect(0, 0, width, height);
        
        // Draw waveform
        oscCtx.beginPath();
        oscCtx.strokeStyle = '#dc2626';
        oscCtx.lineWidth = 1.5;
        oscCtx.shadowBlur = 8;
        oscCtx.shadowColor = '#dc2626';
        
        if (analyser) {
            const dataArray = new Uint8Array(analyser.fftSize);
            analyser.getByteTimeDomainData(dataArray);
            
            const sliceWidth = width / dataArray.length;
            let x = 0;
            
            for (let i = 0; i < dataArray.length; i++) {
                const v = dataArray[i] / 128.0;
                const y = (v * height) / 2;
                
                if (i === 0) {
                    oscCtx.moveTo(x, y);
                } else {
                    oscCtx.lineTo(x, y);
                }
                x += sliceWidth;
            }
        } else {
            // Demo waveform when no audio
            const time = Date.now() / 1000;
            for (let x = 0; x < width; x++) {
                const y = height / 2 + Math.sin(x * 0.05 + time * 3) * (height / 4) * Math.sin(time * 0.5);
                if (x === 0) oscCtx.moveTo(x, y);
                else oscCtx.lineTo(x, y);
            }
        }
        
        oscCtx.stroke();
        oscCtx.shadowBlur = 0;
        
        // Draw center line
        oscCtx.beginPath();
        oscCtx.strokeStyle = 'rgba(220, 38, 38, 0.2)';
        oscCtx.lineWidth = 1;
        oscCtx.moveTo(0, height / 2);
        oscCtx.lineTo(width, height / 2);
        oscCtx.stroke();
        
        oscilloscopeAnimId = requestAnimationFrame(drawOscilloscope);
    }
    
    function stopOscilloscope() {
        if (oscilloscopeAnimId) {
            cancelAnimationFrame(oscilloscopeAnimId);
            oscilloscopeAnimId = null;
        }
        if (oscCtx && oscilloscopeCanvas) {
            oscCtx.clearRect(0, 0, oscilloscopeCanvas.width, oscilloscopeCanvas.height);
        }
    }
