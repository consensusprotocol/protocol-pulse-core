## PHASE 5: MAKE x_service.py ROUTE THROUGH QUALITY GATE

```python
def post_article_tweet(self, article, base_url=''):
    """Route article tweets through unified quality gate."""
    if not self.client and not self.client_v2:
        logging.warning("Twitter API not configured - skipping tweet")
        return None
    try:
        from services.unified_post_gate import unified_post_gate
        result = unified_post_gate.process_article_post(
            article.id,
            article.title,
            getattr(article, 'summary', '') or ''
        )
        if result.get("approved") and result.get("tweet_text"):
            tweet_text = result["tweet_text"][:280]
            if self.client_v2:
                r = self.client_v2.create_tweet(text=tweet_text)
                return str(r.data["id"]) if r and r.data else None
            elif self.client:
                response = self.client.update_status(tweet_text)
                return response.id
        else:
            logging.info(f"Tweet rejected by quality gate: {result.get('rejection_reason', 'unknown')}")
            return None
    except Exception as e:
        logging.error(f"Failed to post tweet: {e}")
        return None
```

## PHASE 6: ADD ARTICLE DEDUPLICATION

Articles with nearly identical titles keep getting published (e.g., three articles all titled “Bitcoin’s Quiet Revolution: Network Resilience…”).

### In services/content_engine.py:

Add this method to the ContentEngine class (before `_save_article_to_db`):

```python
def _is_duplicate(self, title, hours=48):
    """Check if a similar article was published recently."""
    from datetime import datetime, timedelta
    from app import db
    import models
    
    cutoff = datetime.utcnow() - timedelta(hours=hours)
    recent = models.Article.query.filter(
        models.Article.created_at > cutoff,
        models.Article.published == True
    ).all()
    
    new_words = set(title.lower().split())
    
    for article in recent:
        existing_words = set(article.title.lower().split())
        if not existing_words:
            continue
        overlap = len(new_words & existing_words) / max(len(new_words | existing_words), 1)
        if overlap > 0.45:
            return True, article.title
        
        new_start = ' '.join(title.lower().split()[:4])
        existing_start = ' '.join(article.title.lower().split()[:4])
        if new_start == existing_start and len(new_start) > 15:
            return True, article.title
    
    return False, None
```
